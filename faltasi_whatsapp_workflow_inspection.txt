from .models import Organization, OrganizationMember


def is_platform_administrator(user):
    """
    Check whether a user is authorized to administer the platform.
    """

    if not user.is_authenticated:
        return False

    if user.is_superuser:
        return True

    return OrganizationMember.objects.filter(
        user=user,
        is_active=True,
        organization__organization_type="platform_owner",
        organization__status="active",
        role__in=["owner", "admin"],
    ).exists()


def get_user_organization(user, request=None):
    """
    Resolve the organization for the current user.

    Platform administrators:
        Use the organization selected in the session.
        Otherwise, default to the active platform-owner organization.

    Normal users:
        Use their first active organization membership.
    """

    if not user.is_authenticated:
        return None

    if is_platform_administrator(user):

        if request is not None:
            organization_id = request.session.get(
                "active_organization_id"
            )

            if organization_id:
                organization = Organization.objects.filter(
                    id=organization_id,
                    status="active",
                ).first()

                if organization:
                    return organization

                request.session.pop(
                    "active_organization_id",
                    None,
                )

        return Organization.objects.filter(
            organization_type="platform_owner",
            status="active",
        ).order_by("id").first()
    if request is not None:
        organization_id = request.session.get(
            "active_organization_id"
        )

        if organization_id:
            membership = OrganizationMember.objects.filter(
                user=user,
                organization_id=organization_id,
                is_active=True,
                organization__status="active",
            ).select_related("organization").first()

            if membership:
                return membership.organization
            request.session.pop(
                "active_organization_id",
                None,
            )

    membership = (
        OrganizationMember.objects
        .filter(
            user=user,
            is_active=True,
            organization__status="active",
        )
        .select_related("organization")
        .first()
    )

    return membership.organization if membership else None

from django.contrib.auth import update_session_auth_hash
from django.contrib.auth.decorators import login_required
from django.contrib.auth.forms import PasswordChangeForm
from django.shortcuts import render
from organizations.models import OrganizationMember


@login_required
def force_password_change(request):
    """Complete the required first-login password change for provisioned owners."""
    memberships = OrganizationMember.objects.filter(
        user=request.user, is_active=True, must_change_password=True
    )
    if not memberships.exists():
        return redirect("business_requests:business_dashboard")

    form = PasswordChangeForm(request.user, request.POST or None)
    if request.method == "POST" and form.is_valid():
        user = form.save()
        update_session_auth_hash(request, user)
        memberships.update(must_change_password=False)
        messages.success(request, "Your password has been changed successfully.")
        return redirect("business_requests:business_dashboard")
    return render(request, "accounts/force_password_change.html", {"form": form})

@login_required
def conversation_list(request):
    """
    Display all WhatsApp conversations belonging
    to the logged-in organization.
    """

    organization = get_user_organization(request.user)

    if not organization:
        messages.error(
            request,
            "Your account is not connected to an active organization.",
        )
        return redirect("accounts:login")

    conversations = (
        Conversation.objects
        .filter(
            organization=organization,
        )
        .select_related(
            "customer",
            "whatsapp_phone_number",
            "business_request",
        )
        .order_by(
            "-last_message_at",
            "-id",
        )
    )

    search = request.GET.get("search", "").strip()

    if search:
        conversations = conversations.filter(
            customer__name__icontains=search
        ) | conversations.filter(
            phone_number__icontains=search
        )

        conversations = conversations.distinct()

    status = request.GET.get("status", "").strip()

    if status:
        conversations = conversations.filter(
            status=status
        )

    context = {
        "organization": organization,
        "conversations": conversations,
        "search": search,
        "selected_status": status,
        "status_choices": Conversation.STATUS_CHOICES,
    }

    return render(
        request,
        "whatsapp/conversation_list.html",
        context,
    )
